cmake_minimum_required(VERSION 3.20)
project(tiplay_neteq LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
add_compile_options(-ffile-prefix-map=${CMAKE_SOURCE_DIR}=.)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
add_subdirectory(abseil-cpp-20250127.1 absl-build)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_DISABLE_INTRINSICS ON CACHE BOOL "" FORCE)
add_subdirectory(opus-1.5.2 opus-build)
set(W ${CMAKE_CURRENT_SOURCE_DIR}/webrtc)
file(GLOB NETEQ ${W}/modules/audio_coding/neteq/*.cc)
list(FILTER NETEQ EXCLUDE REGEX "(test|mock)")
file(GLOB SPL ${W}/common_audio/signal_processing/*.c)
list(FILTER SPL EXCLUDE REGEX "(_mips|_neon|_sse|_unittest)")
set(EXTRA api/neteq/neteq.cc api/neteq/tick_timer.cc api/neteq/default_neteq_controller_factory.cc api/audio/audio_frame.cc api/audio_codecs/audio_decoder.cc api/audio_codecs/audio_format.cc api/rtp_headers.cc api/rtp_packet_info.cc api/audio_codecs/opus/audio_decoder_opus.cc modules/audio_coding/codecs/opus/audio_decoder_opus.cc modules/audio_coding/codecs/opus/opus_interface.cc modules/audio_coding/codecs/legacy_encoded_audio_frame.cc modules/audio_coding/codecs/pcm16b/audio_decoder_pcm16b.cc modules/audio_coding/codecs/pcm16b/pcm16b.c modules/audio_coding/codecs/cng/webrtc_cng.cc common_audio/vad/vad_core.c common_audio/vad/vad_filterbank.c common_audio/vad/vad_gmm.c common_audio/vad/vad_sp.c common_audio/vad/webrtc_vad.c common_audio/signal_processing/dot_product_with_scale.cc common_audio/third_party/spl_sqrt_floor/spl_sqrt_floor.c rtc_base/ref_counter.cc rtc_base/experiments/struct_parameters_parser.cc rtc_base/checks.cc rtc_base/string_encode.cc rtc_base/strings/string_builder.cc rtc_base/experiments/field_trial_parser.cc rtc_base/numerics/exp_filter.cc system_wrappers/source/metrics.cc)
list(TRANSFORM EXTRA PREPEND "${W}/")
add_executable(neteq port/neteq-wrapper.cpp port/platform.cpp ${NETEQ} ${SPL} ${EXTRA})
target_include_directories(neteq BEFORE PRIVATE port ${W} ${W}/common_audio/signal_processing/include ${CMAKE_CURRENT_SOURCE_DIR}/opus-1.5.2/include)
target_compile_definitions(neteq PRIVATE WEBRTC_POSIX NDEBUG absl_nonnull= absl_nullable= RTC_DISABLE_LOGGING RTC_DISABLE_TRACE_EVENTS RTC_DISABLE_METRICS RTC_DISABLE_CHECK_MSG WEBRTC_OPUS_SUPPORT_120MS_PTIME=1)
target_compile_options(neteq PRIVATE -O2 -fno-exceptions -fno-rtti -ffunction-sections -fdata-sections)
target_link_libraries(neteq PRIVATE opus absl::strings absl::str_format)
target_link_options(neteq PRIVATE -mexec-model=reactor -Wl,--export=neteq_create -Wl,--export=neteq_delete -Wl,--export=neteq_input -Wl,--export=neteq_pcm -Wl,--export=neteq_insert -Wl,--export=neteq_insert_float -Wl,--export=neteq_pull -Wl,--export=neteq_stat -Wl,--export=neteq_duration -Wl,--export=neteq_timestamp -Wl,--gc-sections -Wl,--strip-all -Wl,--initial-memory=33554432 -Wl,--max-memory=33554432 -Wl,-z,stack-size=1048576)
